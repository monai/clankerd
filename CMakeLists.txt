cmake_minimum_required(VERSION 3.30)
project(clankerd LANGUAGES NONE)
set(CLANKERD_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CLANKERD_WORK_DIR "${CMAKE_BINARY_DIR}/work")
set(CLANKERD_DOWNLOAD_DIR "${CLANKERD_SOURCE_DIR}/build/work/downloads" CACHE PATH "Verified input cache")
set(CLANKERD_ARTIFACT_DIR "${CLANKERD_SOURCE_DIR}/build/artifact" CACHE PATH "Published bundle symlink")
set(CLANKERD_INPUTS "${CLANKERD_SOURCE_DIR}/cmake/Inputs.cmake" CACHE FILEPATH "Pinned distribution inputs")
set(CLANKERD_SIGNER rcodesign CACHE FILEPATH "Apple code signer")
option(CLANKERD_HOMEBREW "Discover installed Homebrew host libraries" OFF)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
find_program(CLANKERD_CARGO_EXECUTABLE cargo REQUIRED)
set(CLANKERD_CARGO_TARGET_DIR "${CMAKE_BINARY_DIR}/cargo/build")
include(cmake/HostEnvironment.cmake)
include("${CLANKERD_INPUTS}")
include(cmake/Packages.cmake)
include(cmake/Rust.cmake)
include(cmake/Artifact.cmake)
include(cmake/Checks.cmake)

if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
    add_custom_target(guest-runtime ALL DEPENDS cargo-guest management)
endif()
